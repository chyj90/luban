# 自动洞察测试数据

## 一、数据库建表 SQL

```sql
-- ============================================================
-- 场景 1：退货率异常
-- ============================================================

CREATE TABLE orders (
    order_id INT PRIMARY KEY AUTO_INCREMENT,
    product_line_id INT NOT NULL,
    region_id INT NOT NULL,
    batch_id INT NOT NULL,
    channel_id INT DEFAULT NULL,
    logistics_id INT DEFAULT NULL,
    customer_id INT NOT NULL,
    order_date DATE NOT NULL,
    status VARCHAR(20) NOT NULL DEFAULT 'COMPLETED',
    INDEX idx_product_line (product_line_id),
    INDEX idx_region (region_id),
    INDEX idx_batch (batch_id),
    INDEX idx_channel (channel_id),
    INDEX idx_logistics (logistics_id),
    INDEX idx_order_date (order_date)
);

CREATE TABLE returns (
    return_id INT PRIMARY KEY AUTO_INCREMENT,
    order_id INT NOT NULL,
    return_date DATE NOT NULL,
    return_reason VARCHAR(100) NOT NULL,
    return_qty INT NOT NULL DEFAULT 1,
    FOREIGN KEY (order_id) REFERENCES orders(order_id),
    INDEX idx_return_date (return_date)
);

-- ============================================================
-- 场景 2：产能下降
-- ============================================================

CREATE TABLE production_lines (
    line_id INT PRIMARY KEY AUTO_INCREMENT,
    name VARCHAR(50) NOT NULL,
    daily_capacity INT NOT NULL
);

CREATE TABLE equipment (
    equip_id INT PRIMARY KEY AUTO_INCREMENT,
    name VARCHAR(50) NOT NULL,
    type VARCHAR(30) NOT NULL,
    status VARCHAR(20) NOT NULL DEFAULT 'RUNNING'
);

CREATE TABLE work_orders (
    wo_id INT PRIMARY KEY AUTO_INCREMENT,
    line_id INT NOT NULL,
    equipment_id INT NOT NULL,
    planned_qty INT NOT NULL,
    actual_qty INT NOT NULL,
    work_date DATE NOT NULL,
    FOREIGN KEY (line_id) REFERENCES production_lines(line_id),
    FOREIGN KEY (equipment_id) REFERENCES equipment(equip_id),
    INDEX idx_line_date (line_id, work_date),
    INDEX idx_equip_date (equipment_id, work_date)
);

CREATE TABLE equipment_logs (
    log_id INT PRIMARY KEY AUTO_INCREMENT,
    equip_id INT NOT NULL,
    event_type VARCHAR(20) NOT NULL,
    start_time DATETIME NOT NULL,
    end_time DATETIME,
    FOREIGN KEY (equip_id) REFERENCES equipment(equip_id),
    INDEX idx_equip_time (equip_id, start_time)
);

-- ============================================================
-- 场景 3：客诉上升
-- ============================================================

CREATE TABLE sales_channels (
    channel_id INT PRIMARY KEY AUTO_INCREMENT,
    name VARCHAR(50) NOT NULL,
    type VARCHAR(20) NOT NULL
);

CREATE TABLE logistics (
    logistics_id INT PRIMARY KEY AUTO_INCREMENT,
    name VARCHAR(50) NOT NULL,
    avg_delivery_days DECIMAL(5,2) NOT NULL DEFAULT 3.00
);

CREATE TABLE complaints (
    complaint_id INT PRIMARY KEY AUTO_INCREMENT,
    order_id INT NOT NULL,
    channel_id INT NOT NULL,
    logistics_id INT NOT NULL,
    complaint_type VARCHAR(30) NOT NULL,
    complaint_date DATE NOT NULL,
    INDEX idx_channel_date (channel_id, complaint_date),
    INDEX idx_logistics_date (logistics_id, complaint_date)
);

-- ============================================================
-- 场景 4：库存积压
-- ============================================================

CREATE TABLE materials (
    material_id INT PRIMARY KEY AUTO_INCREMENT,
    name VARCHAR(50) NOT NULL,
    type VARCHAR(20) NOT NULL,
    safety_stock INT NOT NULL
);

CREATE TABLE suppliers (
    supplier_id INT PRIMARY KEY AUTO_INCREMENT,
    name VARCHAR(50) NOT NULL
);

CREATE TABLE inventory (
    inv_id INT PRIMARY KEY AUTO_INCREMENT,
    material_id INT NOT NULL,
    qty INT NOT NULL,
    inbound_date DATE NOT NULL,
    FOREIGN KEY (material_id) REFERENCES materials(material_id),
    INDEX idx_material_date (material_id, inbound_date)
);

CREATE TABLE purchase_orders (
    po_id INT PRIMARY KEY AUTO_INCREMENT,
    material_id INT NOT NULL,
    supplier_id INT NOT NULL,
    qty INT NOT NULL,
    order_date DATE NOT NULL,
    expected_date DATE NOT NULL,
    FOREIGN KEY (material_id) REFERENCES materials(material_id),
    FOREIGN KEY (supplier_id) REFERENCES suppliers(supplier_id),
    INDEX idx_material_date (material_id, order_date)
);

CREATE TABLE production_plans (
    plan_id INT PRIMARY KEY AUTO_INCREMENT,
    material_id INT NOT NULL,
    planned_qty INT NOT NULL,
    plan_date DATE NOT NULL,
    FOREIGN KEY (material_id) REFERENCES materials(material_id),
    INDEX idx_material_date (material_id, plan_date)
);

-- ============================================================
-- 场景 5：成本异常
-- ============================================================

CREATE TABLE cost_items (
    item_id INT PRIMARY KEY AUTO_INCREMENT,
    name VARCHAR(50) NOT NULL,
    type VARCHAR(20) NOT NULL,
    budget_amount DECIMAL(12,2) NOT NULL
);

CREATE TABLE cost_records (
    record_id INT PRIMARY KEY AUTO_INCREMENT,
    item_id INT NOT NULL,
    line_id INT NOT NULL,
    actual_amount DECIMAL(12,2) NOT NULL,
    record_date DATE NOT NULL,
    FOREIGN KEY (item_id) REFERENCES cost_items(item_id),
    INDEX idx_item_date (item_id, record_date),
    INDEX idx_line_date (line_id, record_date)
);

CREATE TABLE raw_materials (
    material_id INT PRIMARY KEY AUTO_INCREMENT,
    name VARCHAR(50) NOT NULL,
    unit_price DECIMAL(10,2) NOT NULL,
    price_date DATE NOT NULL,
    INDEX idx_name_date (name, price_date)
);

CREATE TABLE defect_records (
    record_id INT PRIMARY KEY AUTO_INCREMENT,
    line_id INT NOT NULL,
    wo_id INT NOT NULL,
    produced_qty INT NOT NULL,
    defect_qty INT NOT NULL,
    record_date DATE NOT NULL,
    INDEX idx_line_date (line_id, record_date),
    INDEX idx_wo_date (wo_id, record_date)
);
```

## 五、测试查询方式

测试场景的查询不走 ToolDefinition（平台当前仅支持 HTTP 和 MCP_PASSTHROUGH 类型工具），而是通过 **nl2sql** 直接查询测试数据库。Agent 在 ReAct 循环中生成 SQL，由 `nl2sql_executor` 节点执行，结果返回给 LLM 继续推理。

**各场景对应的 SQL 查询**：

| 场景 | 查询目的 | 示例 SQL |
|------|---------|---------|
| 退货率 | 整体退货率趋势 | `SELECT o.order_date, COUNT(r.return_id)*100.0/COUNT(DISTINCT o.order_id) FROM orders o LEFT JOIN returns r ON o.order_id=r.order_id WHERE o.order_date BETWEEN '2024-08-01' AND '2024-08-31' GROUP BY o.order_date` |
| 退货率 | 按产品线下钻 | `SELECT o.product_line_id, COUNT(r.return_id)*100.0/COUNT(DISTINCT o.order_id) FROM orders o LEFT JOIN returns r ON o.order_id=r.order_id WHERE o.order_date BETWEEN '2024-08-01' AND '2024-08-31' GROUP BY o.product_line_id` |
| 退货率 | 按地区下钻 | `SELECT o.region_id, COUNT(r.return_id)*100.0/COUNT(DISTINCT o.order_id) FROM orders o LEFT JOIN returns r ON o.order_id=r.order_id WHERE o.product_line_id=1 AND o.order_date BETWEEN '2024-08-01' AND '2024-08-31' GROUP BY o.region_id` |
| 退货率 | 按批次下钻 | `SELECT o.batch_id, COUNT(r.return_id)*100.0/COUNT(DISTINCT o.order_id) FROM orders o LEFT JOIN returns r ON o.order_id=r.order_id WHERE o.product_line_id=1 AND o.region_id=1 AND o.order_date BETWEEN '2024-08-01' AND '2024-08-31' GROUP BY o.batch_id` |
| 产能 | 整体产能趋势 | `SELECT work_date, SUM(actual_qty) FROM work_orders WHERE work_date BETWEEN '2024-08-01' AND '2024-08-31' GROUP BY work_date ORDER BY work_date` |
| 产能 | 按产线下钻 | `SELECT line_id, SUM(actual_qty) FROM work_orders WHERE work_date BETWEEN '2024-08-01' AND '2024-08-31' GROUP BY line_id` |
| 产能 | 按工单下钻 | `SELECT wo_id, line_id, equipment_id, planned_qty, actual_qty FROM work_orders WHERE line_id=3 AND work_date BETWEEN '2024-08-01' AND '2024-08-31' ORDER BY work_date` |
| 产能 | 设备日志 | `SELECT * FROM equipment_logs WHERE equip_id=2 AND start_time BETWEEN '2024-08-01' AND '2024-08-31'` |
| 客诉 | 整体客诉率 | `SELECT o.order_date, COUNT(c.complaint_id)*100.0/COUNT(DISTINCT o.order_id) FROM orders o LEFT JOIN complaints c ON o.order_id=c.order_id WHERE o.order_date BETWEEN '2024-08-01' AND '2024-08-31' GROUP BY o.order_date` |
| 客诉 | 按渠道下钻 | `SELECT c.channel_id, COUNT(*) FROM complaints c WHERE c.complaint_date BETWEEN '2024-08-01' AND '2024-08-31' GROUP BY c.channel_id` |
| 客诉 | 按物流商下钻 | `SELECT c.logistics_id, COUNT(*) FROM complaints c WHERE c.channel_id=2 AND c.complaint_date BETWEEN '2024-08-01' AND '2024-08-31' GROUP BY c.logistics_id` |
| 客诉 | 订单量趋势 | `SELECT order_date, COUNT(*) FROM orders WHERE order_date BETWEEN '2024-08-01' AND '2024-08-31' GROUP BY order_date ORDER BY order_date` |
| 库存 | 库存周转 | `SELECT m.name, SUM(i.qty) as total_qty, p.planned_qty FROM materials m JOIN inventory i ON m.material_id=i.material_id JOIN production_plans p ON m.material_id=p.material_id WHERE i.inbound_date <= '2024-08-31' GROUP BY m.name, p.planned_qty` |
| 库存 | 按物料下钻 | `SELECT m.name, i.qty, i.inbound_date FROM inventory i JOIN materials m ON i.material_id=m.material_id WHERE i.inbound_date BETWEEN '2024-08-01' AND '2024-08-31' ORDER BY i.inbound_date` |
| 库存 | 采购订单 | `SELECT po.*, s.name as supplier_name FROM purchase_orders po JOIN suppliers s ON po.supplier_id=s.supplier_id WHERE po.material_id IN (1,2) AND po.order_date BETWEEN '2024-08-01' AND '2024-08-31'` |
| 成本 | 成本偏差 | `SELECT ci.name, ci.budget_amount, SUM(cr.actual_amount) as actual FROM cost_items ci JOIN cost_records cr ON ci.item_id=cr.item_id WHERE cr.record_date BETWEEN '2024-08-01' AND '2024-08-31' GROUP BY ci.name, ci.budget_amount` |
| 成本 | 原料价格 | `SELECT price_date, unit_price FROM raw_materials WHERE material_id=1 AND price_date BETWEEN '2024-07-01' AND '2024-08-31' ORDER BY price_date` |
| 成本 | 废品率 | `SELECT record_date, SUM(defect_qty)*100.0/SUM(produced_qty) FROM defect_records WHERE line_id=1 AND record_date BETWEEN '2024-08-01' AND '2024-08-31' GROUP BY record_date` |
| 专线故障 | 当前活跃告警 | `SELECT a.severity, a.alarm_type, a.description, a.alarm_time, dl.name FROM alarms a JOIN dedicated_lines dl ON a.source_id=dl.line_id WHERE a.severity IN ('CRITICAL','MAJOR') AND a.clear_time IS NULL` |
| 专线故障 | 按光缆段下钻 | `SELECT a.alarm_id, a.alarm_type, a.severity, fs.src_station, fs.dst_station, fs.fiber_length FROM alarms a JOIN fiber_segments fs ON a.source_id=fs.seg_id WHERE a.source_type='FIBER' AND a.severity='CRITICAL' AND a.clear_time IS NULL` |
| 专线故障 | 按传输段下钻 | `SELECT a.alarm_id, a.alarm_type, ts.src_station, ts.dst_station, ts.wavelength FROM alarms a JOIN transmission_segments ts ON a.source_id=ts.seg_id WHERE a.source_type='TRANSMISSION' AND a.severity='CRITICAL' AND a.clear_time IS NULL` |
| 专线故障 | 按 IP 链路下钻 | `SELECT a.alarm_id, a.alarm_type, il.src_station, il.dst_station, il.ip_prefix FROM alarms a JOIN ip_links il ON a.source_id=il.link_id WHERE a.source_type='IP' AND a.clear_time IS NULL` |
| 专线故障 | 承载关系追溯 | `SELECT nt.upper_type, dl.name AS upper_name, nt.lower_type, dl2.name AS lower_name FROM network_topology nt JOIN dedicated_lines dl ON nt.upper_id=dl.line_id JOIN dedicated_lines dl2 ON nt.lower_id=dl2.line_id WHERE nt.relation_type='CARRIED_BY'` |

**审核要点**（超管在 Diff 视图中逐条检查）：

1. 每个概念的 `anomaly_threshold_expr` 是否正确
2. ConceptMapping 的 `table_name` 和 `column_name` 是否匹配实际表结构
3. `computed_expr` 的计算逻辑是否正确（如退货率 = SUM(return_qty)/SUM(order_qty)）
4. ConceptJoinMapping 的 `join_condition` 是否正确（ON 条件字段匹配）
5. ConceptRelation 的 DRILLS_INTO 链是否闭合（根概念 → 末端概念维度可达）
6. CORRELATED 关联关系是否合理（交叉验证维度是否对齐）

### 7.1 数据库建表 SQL

```sql
-- 站点表
CREATE TABLE stations (
    station_id INT PRIMARY KEY,
    name VARCHAR(64) NOT NULL,
    province VARCHAR(32) NOT NULL,
    station_type VARCHAR(32) NOT NULL DEFAULT 'CORE'
);

-- 专线表（OTN二层 / VPN三层）
CREATE TABLE dedicated_lines (
    line_id INT PRIMARY KEY,
    name VARCHAR(128) NOT NULL,
    line_type VARCHAR(16) NOT NULL CHECK (line_type IN ('OTN', 'VPN')),
    src_station INT NOT NULL REFERENCES stations(station_id),
    dst_station INT NOT NULL REFERENCES stations(station_id),
    bandwidth VARCHAR(32),
    status VARCHAR(16) NOT NULL DEFAULT 'ACTIVE'
);

-- 光缆段（物理层，Layer 1）
CREATE TABLE fiber_segments (
    seg_id INT PRIMARY KEY,
    line_id INT NOT NULL REFERENCES dedicated_lines(line_id),
    seg_seq INT NOT NULL,
    src_station INT NOT NULL REFERENCES stations(station_id),
    dst_station INT NOT NULL REFERENCES stations(station_id),
    fiber_length DECIMAL(8,2),
    fiber_type VARCHAR(32) DEFAULT 'G.652',
    core_no VARCHAR(16)
);

-- 传输段（OTN层，Layer 2）
CREATE TABLE transmission_segments (
    seg_id INT PRIMARY KEY,
    otn_line_id INT NOT NULL REFERENCES dedicated_lines(line_id),
    seg_seq INT NOT NULL,
    src_station INT NOT NULL REFERENCES stations(station_id),
    dst_station INT NOT NULL REFERENCES stations(station_id),
    wavelength VARCHAR(16),
    rate VARCHAR(16) DEFAULT '100G'
);

-- IP链路（VPN层，Layer 3）
CREATE TABLE ip_links (
    link_id INT PRIMARY KEY,
    vpn_line_id INT NOT NULL REFERENCES dedicated_lines(line_id),
    link_seq INT NOT NULL,
    src_station INT NOT NULL REFERENCES stations(station_id),
    dst_station INT NOT NULL REFERENCES stations(station_id),
    ip_prefix VARCHAR(32),
    routing_protocol VARCHAR(16) DEFAULT 'BGP',
    as_number INT
);

-- 端口表
CREATE TABLE ports (
    port_id INT PRIMARY KEY,
    device_id INT NOT NULL,
    port_name VARCHAR(64) NOT NULL,
    port_type VARCHAR(32) NOT NULL,
    owner_type VARCHAR(32) NOT NULL CHECK (owner_type IN ('FIBER_SEGMENT', 'TRANSMISSION_SEGMENT', 'IP_LINK')),
    owner_id INT NOT NULL,
    admin_status VARCHAR(16) DEFAULT 'UP',
    oper_status VARCHAR(16) DEFAULT 'UP',
    rx_power DECIMAL(6,2),
    tx_power DECIMAL(6,2)
);

-- 告警表
CREATE TABLE alarms (
    alarm_id INT PRIMARY KEY AUTO_INCREMENT,
    source_type VARCHAR(16) NOT NULL CHECK (source_type IN ('FIBER', 'TRANSMISSION', 'IP')),
    source_id INT NOT NULL,
    severity VARCHAR(16) NOT NULL CHECK (severity IN ('CRITICAL', 'MAJOR', 'MINOR', 'WARNING')),
    alarm_type VARCHAR(64) NOT NULL,
    alarm_time TIMESTAMP NOT NULL,
    clear_time TIMESTAMP NULL,
    description VARCHAR(256),
    INDEX idx_source (source_type, source_id),
    INDEX idx_time (alarm_time),
    INDEX idx_severity (severity)
);

-- 网络拓扑关系表
CREATE TABLE network_topology (
    topo_id INT PRIMARY KEY AUTO_INCREMENT,
    upper_type VARCHAR(32) NOT NULL,
    upper_id INT NOT NULL,
    lower_type VARCHAR(32) NOT NULL,
    lower_id INT NOT NULL,
    relation_type VARCHAR(32) NOT NULL DEFAULT 'CARRIED_BY',
    UNIQUE KEY uk_relation (upper_type, upper_id, lower_type, lower_id, relation_type)
);
```

### 7.4 测试查询方式

测试场景的查询走 **nl2sql** 直接查询测试数据库。

| 查询目的 | 示例 SQL |
|---------|---------|
| 整体告警统计 | `SELECT severity, source_type, COUNT(*) cnt FROM alarms WHERE clear_time IS NULL GROUP BY severity, source_type ORDER BY cnt DESC` |
| 按专线下钻 | `SELECT dl.name, a.severity, a.source_type, COUNT(*) cnt FROM alarms a LEFT JOIN dedicated_lines dl ON (a.source_type='FIBER' AND dl.line_id IN (SELECT line_id FROM fiber_segments WHERE seg_id=a.source_id)) OR (a.source_type='TRANSMISSION' AND dl.line_id IN (SELECT otn_line_id FROM transmission_segments WHERE seg_id=a.source_id)) OR (a.source_type='IP' AND dl.line_id IN (SELECT vpn_line_id FROM ip_links WHERE link_id=a.source_id)) WHERE a.clear_time IS NULL GROUP BY dl.name, a.severity, a.source_type` |
| 按网络层次下钻 | `SELECT source_type, severity, COUNT(*) cnt FROM alarms WHERE clear_time IS NULL GROUP BY source_type, severity ORDER BY FIELD(source_type, 'FIBER', 'TRANSMISSION', 'IP'), FIELD(severity, 'CRITICAL', 'MAJOR', 'MINOR', 'WARNING')` |
| 光缆段告警详情 | `SELECT fs.seg_id, s1.name src, s2.name dst, a.alarm_type, a.severity, a.description, a.alarm_time FROM alarms a JOIN fiber_segments fs ON a.source_id=fs.seg_id JOIN stations s1 ON fs.src_station=s1.station_id JOIN stations s2 ON fs.dst_station=s2.station_id WHERE a.source_type='FIBER' AND a.clear_time IS NULL ORDER BY a.severity, a.alarm_time` |
| 光缆段端口收光功率 | `SELECT p.port_id, p.port_name, p.rx_power, p.tx_power, s1.name src, s2.name dst FROM ports p JOIN fiber_segments fs ON p.owner_id=fs.seg_id AND p.owner_type='FIBER_SEGMENT' JOIN stations s1 ON fs.src_station=s1.station_id JOIN stations s2 ON fs.dst_station=s2.station_id WHERE fs.seg_id=2 ORDER BY p.port_name` |

### 7.5 平台配置指南

#### 概念 1：专线告警（根概念）

| 属性 | 值 |
|------|-----|
| 名称 | 专线告警 |

| 行业 | 运营商 |
| anomaly_threshold_expr | `告警级别 >= MAJOR AND 未清除` |
| anomaly_threshold_desc | 存在MAJOR及以上未清除告警判定为异常 |

**ConceptMapping**：

| 序号 | table_name | column_name | attribute_name | mapping_type | computed_expr |
|------|-----------|-------------|---------------|-------------|--------------|
| 1 | alarms | alarm_id | 告警总数 | computed | `COUNT(alarms.alarm_id)` |
| 2 | alarms | severity | 严重告警数 | computed | `COUNT(CASE WHEN alarms.severity IN ('CRITICAL','MAJOR') THEN 1 END)` |
| 3 | alarms | alarm_time | 告警时间 | direct | - |

#### 概念 2：专线

| 属性 | 值 |
|------|-----|
| 名称 | 专线 |
| 描述 | 跨省专线（OTN二层/VPN三层） |
