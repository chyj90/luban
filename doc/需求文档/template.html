CREATE TABLE IF NOT EXISTS hr_attendance (
id INT AUTO_INCREMENT PRIMARY KEY COMMENT '主键',
employee_no VARCHAR(20) NOT NULL COMMENT '员工ID',
work_days INT COMMENT '出勤天数',
leave_days INT COMMENT '请假天数',
overtime_hours DECIMAL(6,1) COMMENT '加班时长',
late_count INT COMMENT '迟到次数',
month VARCHAR(7) COMMENT '月份(YYYY-MM)',
UNIQUE KEY uk_emp_month (employee_no, month)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='员工考勤表';
